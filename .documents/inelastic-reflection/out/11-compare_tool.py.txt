"""
The comparison of inelastic-reflection prompt 11 (orchestrator note 11, ruling 3), adapted from
jordan-normalization prompt 09's compare_tool.py to alpha. Run from the repository root with
PYTHONPATH=. (it imports Units, for the unit conversions, and nothing that starts Ray or opens a
store):

    PYTHONPATH=. ./venv/bin/python compare_tool.py REF.txt RES.txt TOOLDIR

REF.txt and RES.txt are the stored-values reader's outputs on the two stores
(.documents/inelastic-reflection/measure/histories.py; its JSON lines are read); TOOLDIR holds the
eight outputs of tools/history_and_bbn.py, named <beta>-a<alpha>.txt. Reads only; writes nothing
but its output.

1. The inputs: atol, rtol, the tuning tolerance, T*, pi*, phi*, M, beta, alpha, and z_samples
   against the tool's samples=.
2. Convergence: failure 0, validated 1, abs(residual) < tolerance for both residuals, the
   inequality of ComputeTargets.ScalarHistory.tuning_converged.
3. The bitwise comparison (JN 09's H3): each tool `tuning` line's fifteen values against both
   stores' row: a float by float.hex of the printed repr, an int and the string exactly; a NaN
   printed where the store holds NULL counts as equal and is said.
4. Measured, at alpha > 0: each field of the tool's `dissipation` line against the stored column,
   as float(f"{stored:.6g}") == the printed value (log 10, ruling 7's "parses back"); None against
   NULL. Not part of the verdict's bitwise count; reported.
"""

import json
import math
import re
import sys
from pathlib import Path

from Units import Planck_units

KEYS = (
    "normalization", "V_ref", "phi_ref", "phi_0", "pi_0", "tracking", "residual_H",
    "residual_phi", "tuning_iterations", "plain_steps", "secant_steps", "guard_steps",
    "seed_kind", "seed_normalization", "seed_phi_ref",
)
INTS = {"tuning_iterations", "plain_steps", "secant_steps", "guard_steps"}
STRS = {"seed_kind"}

# the eight histories: (beta, M/M_P, phi*, alpha, tool file)
POINTS = [
    (beta, 1e-3, 5.0, alpha, f"{beta:g}-a{alpha:g}.txt")
    for alpha in (0.0, 0.1)
    for beta in (0.5, 1.5, 3.0, 5.5)
]

DISSIPATION_FIELDS = (
    "injections", "resolved_candidates", "unresolved_candidates", "injected_fraction",
    "rho_chi_ratio_1MeV", "rho_chi_ratio_70keV", "rho_chi_ratio_today", "Delta_N_eff_1MeV",
    "Delta_N_eff_today", "log_R_first_impact", "parking_N", "parking_log_T_Jordan_GeV",
    "release_N", "release_log_T_Jordan_GeV", "worst_birth_margin", "worst_birth_margin_N",
    "birth_failing_share", "worst_reflection_margin", "worst_reflection_margin_N",
    "reflection_failing_share", "worst_tracker_margin", "worst_tracker_margin_N",
    "tracker_failing_share", "unassessed_share",
)

def rows(path):
    out = []
    for line in Path(path).read_text().splitlines():
        if line.startswith("JSON "):
            out.append(json.loads(line[5:]))
    return out


def fval(rec, key):
    v = rec[key]
    if isinstance(v, dict):
        return float.fromhex(v["hex"])
    return v


def find(recs, beta, M_eV, phi, alpha):
    hits = [
        r for r in recs
        if fval(r, "beta") == beta and fval(r, "phi_init_Mp") == phi and fval(r, "M_value_eV") == M_eV
        and fval(r, "alpha") == alpha
    ]
    return hits


def tool_lines(path):
    text = Path(path).read_text()
    tuning = re.search(r"^tuning [^:]*: (.*)$", text, re.M).group(1)
    vals = dict(kv.split("=", 1) for kv in tuning.split(" "))
    samples = int(re.search(r"^history [^:]*: .* samples=(\d+) ", text, re.M).group(1))
    hist = re.search(r"^history .*$", text, re.M).group(0)
    bbn = re.search(r"^bbn .*$", text, re.M).group(0)
    up = re.search(r"^uptime: .*load averages: (.*)$", text, re.M).group(1)
    m = re.search(r"^dissipation [^:]*: (.*)$", text, re.M)
    diss = dict(kv.split("=", 1) for kv in m.group(1).split(" ")) if m else None
    return vals, samples, hist, bbn, up, diss


def main(argv):
    ref, res = rows(argv[0]), rows(argv[1])
    tooldir = Path(argv[2])
    units = Planck_units()
    stores = (("reference", ref), ("resumed", res))
    T_hex = (2.0e4 * units.GeV / units.GeV).hex()
    print(f"T* as the build stores it: (2e4 * GeV) / GeV = {2.0e4 * units.GeV / units.GeV!r} ({T_hex}); "
          f"2e4 * GeV = {2.0e4 * units.GeV!r} ({(2.0e4 * units.GeV).hex()})")
    print(f"rows: reference {len(ref)}, resumed {len(res)}")
    all_equal = True
    all_inputs = True
    all_conv = True
    diss_all = True
    for beta, M, phi, alpha, tf in POINTS:
        M_eV = (M * units.PlanckMass) / units.eV
        vals, samples, hist, bbn, up, diss = tool_lines(tooldir / tf)
        print()
        print(f"=== beta={beta:g} M={M:g} M_P phi*={phi:g} alpha={alpha:g}  (tool {tf}; load at its start {up})")
        print(f"    {hist}")
        print(f"    {bbn[:220]}")
        for name, recs in stores:
            hits = find(recs, beta, M_eV, phi, alpha)
            if len(hits) != 1:
                print(f"  {name}: {len(hits)} rows match; STOP")
                all_equal = False
                continue
            r = hits[0]
            # 1. inputs
            ins = {
                "log10_atol": fval(r, "log10_atol") == -8.0,
                "log10_rtol": fval(r, "log10_rtol") == -8.0,
                "log10_tuning_tol": fval(r, "log10_tuning_tol") == -5.0,
                "T*": fval(r, "T_init_GeV").hex() == T_hex
                and (fval(r, "T_init_GeV") * units.GeV).hex() == (2.0e4 * units.GeV).hex(),
                "pi*": fval(r, "pi_init_Mp") == 0.0,
                "phi*": fval(r, "phi_init_Mp") == phi,
                "M": fval(r, "M_value_eV").hex() == M_eV.hex(),
                "beta": fval(r, "beta") == beta,
                "alpha": fval(r, "alpha") == alpha,
                "n": r["n_value"] == 1,
                "z_samples": r["z_samples"] == samples,
            }
            ok_in = all(ins.values())
            all_inputs &= ok_in
            # 2. convergence
            tol = 10.0 ** fval(r, "log10_tuning_tol")
            rH, rP = fval(r, "residual_H"), fval(r, "residual_phi")
            conv = r["failure"] == 0 and r["validated"] == 1 and abs(rH) < tol and abs(rP) < tol
            all_conv &= conv
            # 3. bitwise
            diffs, nan_null = [], []
            for k in KEYS:
                printed = vals[k]
                stored = fval(r, k)
                if k in INTS:
                    same = stored == int(printed)
                elif k in STRS:
                    same = stored == printed
                else:
                    p = float(printed)
                    if math.isnan(p):
                        same = stored is None
                        if same:
                            nan_null.append(k)
                    else:
                        same = stored is not None and stored.hex() == p.hex()
                if not same:
                    diffs.append(f"{k}: tool {printed} stored {r[k]}")
            all_equal &= not diffs
            print(f"  {name}: inputs {'OK' if ok_in else 'DIFFER ' + str([k for k, v in ins.items() if not v])} "
                  f"(z_samples {r['z_samples']} vs samples={samples}); converged {conv} "
                  f"(|res_H| {abs(rH):.3e}, |res_phi| {abs(rP):.3e} < {tol:g}); "
                  f"15 keys: {'all equal' if not diffs else 'DIFFER: ' + '; '.join(diffs)}"
                  + (f"; NaN printed / NULL stored: {nan_null}" if nan_null else ""))
            # 4. the dissipation line against the stored columns (measured)
            if diss is not None:
                dd = []
                for k in DISSIPATION_FIELDS:
                    col = k[:-4] if k.endswith("_GeV") else k
                    printed, stored = diss[k], fval(r, col)
                    if printed == "None":
                        ok_k = stored is None
                    else:
                        ok_k = stored is not None and float(f"{stored:.6g}") == float(printed)
                    if not ok_k:
                        dd.append(f"{k}: tool {printed} stored {stored!r}")
                diss_all &= not dd
                print(f"  {name}: dissipation line, 24 fields at :.6g: {'all equal' if not dd else 'DIFFER: ' + '; '.join(dd)}")
        # the per-key table, from the reference row
        r = find(ref, beta, M_eV, phi, alpha)[0]
        print("    key-by-key (tool's printed repr | its float.hex | reference stored hex | resumed stored hex):")
        s = find(res, beta, M_eV, phi, alpha)[0]
        for k in KEYS:
            p = vals[k]
            if k in INTS or k in STRS:
                print(f"      {k:19s} {p} | - | {r[k]} | {s[k]}")
            else:
                print(f"      {k:19s} {p} | {float(p).hex()} | {r[k]['hex'] if isinstance(r[k], dict) else r[k]} | "
                      f"{s[k]['hex'] if isinstance(s[k], dict) else s[k]}")
    print()
    print(f"SUMMARY: inputs {'all as the tool' if all_inputs else 'DIFFER'}; converged {'all' if all_conv else 'NOT ALL'}; "
          f"bitwise {'all 8 x 15 x 2 equal' if all_equal else 'DIFFERENCES FOUND'}; "
          f"dissipation lines (measured, :.6g) {'all equal' if diss_all else 'DIFFER'}")
    return 0 if (all_equal and all_inputs and all_conv) else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
