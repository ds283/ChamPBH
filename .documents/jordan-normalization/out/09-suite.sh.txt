#!/bin/sh
# jordan-normalization 09: run one CLAUDE.md suite command from the repository root, its output to
# var/jordan-normalization-09/suites/<point>-<name>.txt, and a summary (the command, uptime before,
# the wall time, Ran, the verdict, the exit) appended to var/jordan-normalization-09/09-suites.txt.
#   sh var/jordan-normalization-09/tools/suite.sh POINT NAME DIR [PATTERN]
L=var/jordan-normalization-09
point=$1; name=$2; dir=$3; pat=$4
out=$L/suites/$point-$name.txt
if [ -n "$pat" ]; then
    cmd="PYTHONPATH=. ./venv/bin/python -m unittest discover -s $dir -t . -p '$pat'"
else
    cmd="PYTHONPATH=. ./venv/bin/python -m unittest discover -s $dir -t ."
fi
{
    echo "## $point $name: $(date '+%F %T %z')"
    echo "\$ $cmd"
    echo "uptime: $(uptime)"
} >> $L/09-suites.txt
t0=$(date +%s)
if [ -n "$pat" ]; then
    PYTHONPATH=. ./venv/bin/python -m unittest discover -s "$dir" -t . -p "$pat" > "$out" 2>&1
else
    PYTHONPATH=. ./venv/bin/python -m unittest discover -s "$dir" -t . > "$out" 2>&1
fi
rc=$?
t1=$(date +%s)
{
    echo "wall: $((t1 - t0)) s"
    grep -E '^Ran [0-9]+ tests? in' "$out"
    echo "verdict: $(grep -E '^(OK|FAILED)' "$out" | tail -1)"
    echo "exit: $rc"
    echo
} >> $L/09-suites.txt
tail -6 $L/09-suites.txt
